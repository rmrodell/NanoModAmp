Step 1b: Create folder with links to bam file so you only have to run BIDdetect once.

# 1. Define the source directories containing your BAM files.
SOURCE_DIRS=(
    "/scratch/users/rodell/20251106_pool1/invitro/junedata/deduplicated_bam"
    "/scratch/users/rodell/20251106_pool1/invitro/maydata/deduplicated_bam"
)
# 2. Define the target directory where symbolic links will be created.
LINK_DIR="/scratch/users/rodell/20251106_pool1/invitro/bam_links"
# 3. Loop through each source directory
for dir in "${SOURCE_DIRS[@]}"; do
    echo "  -> Processing directory: ${dir}"
    # Use 'find' to locate all .bam and .bai files and link them.
    find "$dir" -type f \( -name "*.bam" -o -name "*.bai" \) -print -exec ln -s {} "$LINK_DIR" \;
done
